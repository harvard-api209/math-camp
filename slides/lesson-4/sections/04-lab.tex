\section{Prepare the lab}

\begin{frame}{Choose one question for Lab 4}
\small
\begin{enumerate}
\item \textbf{Health.} How is income associated with under-five mortality in 2022, and how does the pattern differ by income group?
\item \textbf{Gender.} Where is girls' secondary enrollment associated with lower adolescent fertility in 2022, and where does the pattern depart from the overall comparison?
\item \textbf{Access.} Which economies closed the combined electricity and internet access gap fastest from 2000 to 2022?
\item \textbf{Climate.} Which economies expanded renewable electricity, reduced carbon intensity, and ended with higher real income from 2000 to 2021?
\end{enumerate}
\end{frame}

\begin{frame}{Hints for the 2022 cross-sections}
\textbf{Health}
\begin{itemize}
\item \textbf{Table:} summarise income and mortality within current income groups.
\item \textbf{Figure:} plot GDP per capita against mortality; map income group to color and consider a logarithmic income scale.
\end{itemize}
\vspace{0.3cm}
\textbf{Gender}
\begin{itemize}
\item \textbf{Table:} compare median enrollment and fertility across regions.
\item \textbf{Figure:} plot girls' enrollment against adolescent fertility and map region to color.
\end{itemize}
\vspace{0.25cm}
In both cases, one row is one economy observed in 2022.
\end{frame}

\begin{frame}{Hints for the endpoint comparisons}
\textbf{Access}
\begin{itemize}
\item \textbf{Table:} rank economies by combined electricity and internet gap closure while retaining both components.
\item \textbf{Figure:} plot the two percentage-point changes against one another; add zero reference lines and map region to color.
\end{itemize}
\vspace{0.3cm}
\textbf{Climate}
\begin{itemize}
\item \textbf{Table:} identify economies satisfying all three endpoint conditions and summarise them by region.
\item \textbf{Figure:} plot renewable-electricity change against carbon-intensity change; map income change to color and add zero reference lines.
\end{itemize}
\end{frame}
