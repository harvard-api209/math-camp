\section{Build and record the table}
\subsection{One common path}

\begin{frame}[fragile]{One R script follows the entire lesson}
  Open \texttt{code/lesson-2-walkthrough.R} in RStudio.

  \begin{description}
    \item[Move with the deck] Each numbered section matches the teaching sequence.
    \item[Run one step] Select a section and use Command+Enter on macOS or
      Ctrl+Enter on Windows.
    \item[Inspect objects] Use the Console and Environment panes after each verb.
    \item[Rebuild everything] Use Source to run the complete script from the
      protected input through independent verification.
  \end{description}

\begin{lstlisting}
Rscript code/lesson-2-walkthrough.R
\end{lstlisting}

  The script uses the same names, examples, checks, and output paths as the deck.
\end{frame}

\begin{frame}[fragile]{One complete health pipeline}
\begin{lstlisting}[language=R]
health_analysis <- wdi |>
  select(
    iso3c, country, region, income_level_current, year,
    gdp_per_capita_ppp, under5_mortality
  ) |>
  filter(
    between(year, 2000L, 2022L),
    !is.na(gdp_per_capita_ppp),
    gdp_per_capita_ppp > 0,
    !is.na(under5_mortality)
  ) |>
  mutate(log_income = log(gdp_per_capita_ppp)) |>
  arrange(iso3c, year)

stopifnot(!anyDuplicated(health_analysis[c("iso3c", "year")]))
\end{lstlisting}
\end{frame}

\begin{frame}{Read the pipeline as decisions}
  \begin{enumerate}
    \item \textbf{Population:} economy--years without both measures leave the sample.
    \item \textbf{Scope:} selected columns define the health question.
    \item \textbf{Transformation:} logged income stays beside the original measure.
    \item \textbf{Structure:} the country--year key is tested in code.
  \end{enumerate}

  \vfill
  Each important line should correspond to one decision in the table plan.
\end{frame}

\begin{frame}{Use current metadata within its time meaning}
  \texttt{income\_level\_current} is the classification attached when this
  snapshot was assembled.

  \vspace{0.6em}
  It is useful for a current grouping. It does \alert{not} prove that a country
  belonged to that income group in 2000.

  \vspace{0.6em}
  A historical classification needs a separate country--year source, a retrieval
  date, and documentation.
\end{frame}

\begin{frame}{The build record}
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
      \begin{block}{Rows and key}
        Input and output rows; country--year uniqueness.
      \end{block}
      \begin{block}{Years}
        First year, last year, and coverage.
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{Missingness}
        Observed and missing values by variable.
      \end{block}
      \begin{block}{Ranges}
        Plausible values in documented units.
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{Matches}
        Unmatched and multiplied join keys.
      \end{block}
      \begin{block}{Decision note}
        What changed and why it belongs.
      \end{block}
  \end{columns}

  \vfill
  The record is useful when every check points back to the table plan.
\end{frame}

\begin{frame}[fragile]{Run the saved workflow from a clean process}
  In the RStudio Terminal, not from a console full of objects:

\begin{lstlisting}
Rscript code/02-build-analysis.R
\end{lstlisting}

  It creates two separate artifacts:
  \begin{itemize}
    \item \texttt{outputs/02-health-analysis.csv}: the analysis table;
    \item \texttt{outputs/02-health-build-record.csv}: evidence about how the
      table meets its stated requirements.
  \end{itemize}

  \vfill
  Successful execution is necessary, but it is not yet independent evidence
  that the saved object is correct.
\end{frame}

\begin{frame}[fragile]{Verify the saved object independently}
  Run a second script that reads the frozen source and the saved table afresh:

\begin{lstlisting}
Rscript code/02-verify-analysis.R
\end{lstlisting}

  The verifier checks columns, expected rows, key uniqueness, period, observed
  values, plausible signs, and whether \texttt{log\_income} recomputes from the
  original measure.

  \begin{block}{The visible completion signal}
    \texttt{RESULT: PASS. The saved health table meets its stated requirements.}
  \end{block}

  The verifier is separate so it does not merely trust objects left behind by
  the builder.
\end{frame}

\begin{frame}{Four levels of verification}
  \begin{enumerate}
    \item \textbf{Execution:} Did the process finish and create the named files?
    \item \textbf{Structure:} Do columns, rows, unit, key, and period match the table plan?
    \item \textbf{Values:} Do missingness, ranges, and recomputed variables follow
      their documented definitions?
    \item \textbf{Interpretation:} Does the policy statement respect the sample
      limits and the associational design?
  \end{enumerate}

  \vfill
  An agent can help at every level; it cannot collapse all four into ``the code
  ran.''
\end{frame}
